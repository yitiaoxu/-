        if (job_cnt - result_cnt < numThreads) {

每次读取之前需要重置偏移地址和擦除缓冲区

```
            ioctl(fd, PCI_MAP_ADDR_CMD, &dma_operation);//告诉DMA向DDR请求数据并且把数据存储到BAR0
            for (int i = 0; i < height; i++) //逐行进行传输
            {
                memset(dma_operation.data.read_buf, 0, DMA_MAX_PACKET_SIZE);//擦除读缓存
                ioctl(fd, PCI_DMA_WRITE_CMD, &dma_operation);//触发一次 DMA 传输动作（按你这段流程是“先发起搬运”这一步）
                for (volatile int k = 0; k < 2500; k++); 
                ioctl(fd, PCI_READ_FROM_KERNEL_CMD, &dma_operation);//从缓存中读取数据
                memcpy(image_buf_temp.data() + i * 2560, dma_operation.data.read_buf, 2560);//把当前数据拷贝到缓存对应的位置，拷贝到下一行应该在的位置，避免重叠，逐行拼接完整的 1280×720 原始帧数据
            }
            ioctl(fd, PCI_UMAP_ADDR_CMD, &dma_operation);//本轮 DMA 读取结束后的收尾清理
```
