```
InferenceThread::InferenceThread(QObject *parent)
    : QThread(parent), keep_running(false), current_input_type(INPUT_CAMERA),

{
    std::string yolo_model_path = "./weights/yolov11_3568.rknn";
    std::string lprnet_model_path = "./weights/lprnet_rk3568.rknn";
    int numThreads = 4;

    yolo11_detector = new Yolov11Custom(); //创建一个YOLOv11检测器对象，后面用来加载模型并做检测
    if (yolo11_detector->LoadModel(yolo_model_path.c_str()) == NN_SUCCESS)//尝试从yolo_model_path下加载YOLO模型，只有返回成功才会进入if
    {
        yolo11_thread_pool = new Yolov11ThreadPool();//创建YOLO推理线程对象，用于并行处理，提高推理吞吐
        yolo11_thread_pool->setUp(yolo_model_path, numThreads);//初始化线程池：传入模型路径和线程数numThreads
    }

    memset(&lprnet_ctx, 0, sizeof(rknn_app_context_t));//把lprnet_ctx清0
    if (init_lprnet_model(lprnet_model_path.c_str(), &lprnet_ctx) == 0)//加载LPRNet 模型并把上下文写入 lprnet_ctx。
    {

```

`memset` 是 C/C++ 里用来按字节填充内存的函数。你这行代码：

```
memset(&lprnet_ctx, 0, sizeof(rknn_app_context_t));
```

含义是把 `lprnet_ctx` 这块内存的每个字节都置为 `0`，大小是 `rknn_app_context_t` 结构体的总字节数。

`init_lprnet_model(const char *model_path, rknn_app_context_t *app_ctx)`，该函数的参数解析：`model_path`：模型文件路径（`.rknn`）、`app_ctx`：输出参数，函数会把 RKNN 上下文、输入输出属性、模型尺寸等写进去、返回值：`0` 成功，`-1` 失败。

**model_path==lprnet_model_path.c_str()**

**rknn_app_context_t *app_ctx==&lprnet_ctx**

```
int init_lprnet_model(const char *model_path, rknn_app_context_t *app_ctx) {
    int ret, model_len = 0;
    unsigned char *model_data = load_model_data(model_path, &model_len);
 	//
 		load_model_data(model_path, &model_len)：按 model_path 打开并读取 .rknn 模型文件到堆内存，返回这块内存首地址给 model_data，同时通过 &model_len 回传文件字节大小。
 	//
    if (model_data == NULL) return -1;//模型数据读取失败就返回 -1

    rknn_context ctx = 0;
    ret = rknn_init(&ctx, model_data, model_len, 0, NULL);
    //
    用内存中的 .rknn 模型创建 RKNN 运行时上下文存在ctx下
    //
    free(model_data);//释放之前 load_model_data() 用 malloc 申请的模型内存
    if (ret < 0) return -1;

    rknn_input_output_num io_num;//声明一个变量来保存模型输入输出数量
    rknn_query(ctx, RKNN_QUERY_IN_OUT_NUM, &io_num, sizeof(io_num));
    //
    	rknn_query()可理解为RK的通用信息查询函数,，这个地方用来查询模型输入输出个数，保存到io_num的结构体中
    //
    rknn_tensor_attr input_attrs[io_num.n_input];
    memset(input_attrs, 0, sizeof(input_attrs));
    for (int i = 0; i < io_num.n_input; i++) {
        input_attrs[i].index = i;
        rknn_query(ctx, RKNN_QUERY_INPUT_ATTR, &(input_attrs[i]), sizeof(rknn_tensor_attr));
    }
	//
		先创建一个 input_attrs 数组，然后把每个输入节点的 RKNN 属性写进去。所以最终 input_attrs 里会保存所有输出 tensor 的描述信息（如维度、数据类型、量化参数、格式等）
	//
    rknn_tensor_attr output_attrs[io_num.n_output];
    memset(output_attrs, 0, sizeof(output_attrs));
    for (int i = 0; i < io_num.n_output; i++) {
        output_attrs[i].index = i;
        rknn_query(ctx, RKNN_QUERY_OUTPUT_ATTR, &(output_attrs[i]), sizeof(rknn_tensor_attr));
    }
	//
		先创建一个 output_attrs 数组，然后把每个输出节点的 RKNN 属性写进去。所以最终 output_attrs 里会保存所有输出 tensor 的描述信息（如维度、数据类型、量化参数、格式等）
	//
    app_ctx->rknn_ctx = ctx;
    app_ctx->io_num = io_num;
    app_ctx->input_attrs = (rknn_tensor_attr *)malloc(io_num.n_input * sizeof(rknn_tensor_attr));
    memcpy(app_ctx->input_attrs, input_attrs, io_num.n_input * sizeof(rknn_tensor_attr));
    app_ctx->output_attrs = (rknn_tensor_attr *)malloc(io_num.n_output * sizeof(rknn_tensor_attr));
    memcpy(app_ctx->output_attrs, output_attrs, io_num.n_output * sizeof(rknn_tensor_attr));
    //
    	把函数栈上的临时信息“转存”到堆内存并挂到 app_ctx，避免函数返回后数据失效。
    //
    if (input_attrs[0].fmt == RKNN_TENSOR_NCHW) {
        app_ctx->model_channel = input_attrs[0].dims[1];
        app_ctx->model_height = input_attrs[0].dims[2];
        app_ctx->model_width = input_attrs[0].dims[3];
    } else {
        app_ctx->model_height = input_attrs[0].dims[1];
        app_ctx->model_width = input_attrs[0].dims[2];
        app_ctx->model_channel = input_attrs[0].dims[3];
    }
    //
    	不管模型是 NCHW 还是 NHWC，都把 app_ctx->model_channel / model_height / model_width 填成统一语义
    //
    return 0;
}
```

`init_lprnet_model()` 主要就是两件事：

- 加载并初始化模型运行环境
  读取 `.rknn` 文件到内存，调用 `rknn_init` 得到 `rknn_ctx`。
- 查询并保存模型元信息到 `app_ctx`
  包括输入输出数量、每个输入/输出 tensor 的属性、以及模型输入的 `channel/height/width`。

```
                //
                int new_x = std::max(0, cx - new_w / 2);
                int new_y = std::max(0, cy - new_h / 2);
                safe_box = cv::Rect(new_x, new_y, std::min(draw_frame.cols - new_x, new_w), std::min(draw_frame.rows - new_y, new_h));
                //
                	重新计算左上角 new_x/new_y，并再次用边界裁剪防止扩框后超出图像范围，最终得到安全
```

```
QString InferenceThread::detectPlateColor(const cv::Mat& plate_img) {
    if (plate_img.empty()) return "Unknown";
    cv::Mat hsv, mask_b, mask_y, mask_g;
    cv::cvtColor(plate_img, hsv, cv::COLOR_BGR2HSV);
    cv::inRange(hsv, cv::Scalar(100, 50, 50), cv::Scalar(124, 255, 255), mask_b);
    cv::inRange(hsv, cv::Scalar(15, 50, 50), cv::Scalar(34, 255, 255), mask_y);
    cv::inRange(hsv, cv::Scalar(35, 50, 50), cv::Scalar(85, 255, 255), mask_g);

    int b_pts = cv::countNonZero(mask_b);
    int y_pts = cv::countNonZero(mask_y);
    int g_pts = cv::countNonZero(mask_g);
    int max_pts = std::max({b_pts, y_pts, g_pts});
    
    if (max_pts < 50) return "Unknown";
    if (max_pts == b_pts) return "蓝色";
    if (max_pts == y_pts) return "黄色";
    return "绿色";
}
```

实现路径：<span style='color:red;'>HSV 颜色阈值分割 + 像素计数投票</span>

```
QString InferenceThread::recognizePlateText(const cv::Mat& plate_img)
{
    if (!lprnet_ready || plate_img.empty()) return "识别失败";//模型没初始化好或图像为空，会返回识别失败
    cv::Mat resized;
    cv::resize(plate_img, resized, cv::Size(MODEL_WIDTH, MODEL_HEIGHT));//把车牌图缩放到 LPRNet 固定输入尺寸
    
    lprnet_result result;//创建结果结构体，用来接收识别出的车牌号字符串。
    if (inference_lprnet_model(&lprnet_ctx, resized, &result) != 0) return "推理错误";//调用真正的 RKNN 推理接口
    return QString::fromStdString(result.plate_name);//推理成功，把 std::string 转成 Qt 的 QString 返回给上层 UI/绘制逻辑。
}
```

<span style='color:red;'>RKNN推理接口</span>

```
int inference_lprnet_model(rknn_app_context_t *app_ctx, const cv::Mat& src_img, lprnet_result *out_result) {
    int ret;
    rknn_input inputs[1];
    rknn_output outputs[1];
    memset(inputs, 0, sizeof(inputs));
    memset(outputs, 0, sizeof(outputs));

    inputs[0].index = 0;//指定这是第0号输入张量
    inputs[0].type = RKNN_TENSOR_UINT8;//输入数据类型是uint8
    inputs[0].fmt = RKNN_TENSOR_NHWC;//输入内存排布按NHWC解释
    inputs[0].size = app_ctx->model_width * app_ctx->model_height * app_ctx->model_channel;//输入字节大小 = width * height * channel
    inputs[0].buf = (void*)src_img.data;//输入数据指向 OpenCV 图像内存

    ret = rknn_inputs_set(app_ctx->rknn_ctx, 1, inputs);//把上述输入信息提交给 RKNN 运行
    if (ret < 0) return -1;

    ret = rknn_run(app_ctx->rknn_ctx, nullptr);//触发模型推理
    if (ret < 0) return -1;

    outputs[0].want_float = 1;//要求取出的输出转成 float
    ret = rknn_outputs_get(app_ctx->rknn_ctx, 1, outputs, NULL);//获取第0个输出结果
    if (ret < 0) return ret;

    std::vector<int> no_repeat_blank_label{};
    float prebs[OUT_COLS];//保存每个时间步（每一列）的预测类别索引
    int pre_c;
    for (int x = 0; x < OUT_COLS; x++) //遍历输出的每一列
    {
        float *ptr = (float *)outputs[0].buf;//取模型输出缓冲区首地址（float 数据）
        float preb[OUT_ROWS];
        for (int y = 0; y < OUT_ROWS; y++) //把第 x 列在所有类别 y 上的分数提取出来
        {
            preb[y] = ptr[x];
            ptr += OUT_COLS;
        }
        int max_num_index = std::max_element(preb, preb + OUT_ROWS) - preb;//找到该列分数最大的类别索引（argmax）
        prebs[x] = max_num_index;//把这一列的“最可能字符类别”记录下来
    }
//CTC风格后处理，考虑是否去掉，CTC会将重复字符去掉，可能会影响验证的结果
    pre_c = prebs[0];
    if (pre_c != OUT_ROWS - 1) no_repeat_blank_label.push_back(pre_c);
    
    for (int value : prebs) {
        if (value == OUT_ROWS - 1 || value == pre_c) continue;
        no_repeat_blank_label.push_back(value);
        pre_c = value;
    }

    out_result->plate_name.clear();//先清空结果字符串，避免残留上一次识别内容
    for (int hh : no_repeat_blank_label) //遍历前面得到的类别 ID 序列，把每个 ID 通过 plate_code 映射成字符（省份/字母/数字等），逐个拼接成完整车牌号。
    {
        out_result->plate_name += plate_code[hh];
    }

    rknn_outputs_release(app_ctx->rknn_ctx, 1, outputs);//释放 rknn_outputs_get 获取到的输出缓冲，防止内存/资源泄漏
    return 0;
}
```
